numbers={10,20,10,30,20}
print("How the set behaves with the duplicate values")
print(numbers)
print("Add the element to the set")
a=int(input("Enter a number:"))
numbers.add(a)
print(numbers)
print("Remove an element")
b=int(input("Enter a number:"))
numbers.remove(b)
print(numbers)
print("Using discard")
c=int(input("Enter a number:")
numbers.discard(c)
print(numbers)

set1={10,20,30,40,50}
