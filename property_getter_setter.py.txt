class Student:
    def __init__(self,name):
        self.__name=name
    @property
    def name(self):
        return self.__name
    @name.setter
    def name(self,name):
        self.__name=name
student1=Student("Yasaswini")
print("Name:",student1.name)
student1.name="Rahul"
print("Name after updation:",student1.name)
