class Animal:
    def sound(self):
        print("Animal makes a sound")
class Dog(Animal):
    def bark(self):
        print("Dog barks")
class Cat(Animal):
    def meow(self):
        print("Cat says meow")
        
dog1=Dog()
cat1=Cat()
cat1.sound()
cat1.meow()
dog1.sound()
dog1.bark()
