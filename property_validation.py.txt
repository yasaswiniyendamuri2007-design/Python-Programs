class Student:
    def __init__(self,marks):
        self.__marks=marks
    @property
    def marks(self):
        return self.__marks
    @marks.setter
    def marks(self,marks):
        if 0<=marks<=100:
            self.__marks=marks
            print("Marks updated successfully")
        else:
            print("Invalid marks")
student1=Student(80)
print("Marks:",student1.marks)
student1.marks=95
print("Marks after updation:",student1.marks)
student1.marks=120
print("Marks after updation:",student1.marks)
