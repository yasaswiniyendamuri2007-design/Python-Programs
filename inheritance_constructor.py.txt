class Person:
    def __init__(self,name):
        self.name=name
    def display(self):
        print("Name:",self.name)
class Student(Person):
    def __init__(self,name,course):
        super().__init__(name)
        self.course=course
    def display(self):
        super().display()
        print("Course:",self.course)
student1=Student("Yasaswini","CSE")
student1.display()
