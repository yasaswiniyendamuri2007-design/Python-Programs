class Temperature:
    def __init__(self,celesius):
        self.__celesius=celesius
    @property
    def celesius(self):
        return self.__celesius
    @celesius.setter
    def celesius(self,celesius):
        if celesius>=-273.5:
            self.__celesius=celesius
            print("Celesius updated successfully")
        else:
            print("Invalid celesius")
