class user:
    def __init__(self,password):
        self.__password=password
    @property
    def password(self):
        return self.__password
    @password.setter
    def password(self,password):
        if len(password)>8:
            self.__password=password
            print("Password updated successfully")
        else:
            print("Invalid password")
