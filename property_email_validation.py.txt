class User:
    def __init__(self,email):
        self.__email=email
    @property
    def email(self):
        return self.__email
    @email.setter
    def email(self,email):
        if "@" in email:
            self.__email=email
            print("Email updated successfully")
        else:
            print("Invalid email")
