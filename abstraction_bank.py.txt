from abc import ABC,abstractmethod
class Bank(ABC):
    @abstractmethod
    def interest_rate(self):
        pass
class SBI(Bank):
    def interest_rate(self):
        print("SBI interest rate is 7%")
class HDFC(Bank):
    def interest_rate(self):
        print("HDFC interest rate is 7.5%")
class ICICI(Bank):
    def interest_rate(self):
        print("ICICI interest rate is 7.25%")
sbi1=SBI()
hdfc1=HDFC()
icici1=ICICI()
sbi1.interest_rate()
hdfc1.interest_rate()
icici1.interest_rate()
