import time
def timing_decorator(original_function):
    def wrapper():
        start_time=time.time()
        original_function()
        end_time=time.time()
        print("Execution time",end_time-start_time,"seconds")
    return wrapper
@timing_decorator
def greet():
    time.sleep(2)
    print("Hello,Yasaswini")
greet()
