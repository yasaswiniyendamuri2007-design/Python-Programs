numbers=(9,5,1,2,6,3,8,7)
print(numbers)
print("To print numbers individually")
for i in numbers:
    print(i)
print("Indexing topic")
print(numbers[0])#first element
print(numbers[3])#fourth element
print(numbers[7])#last element
print("Length of numbers")
print(len(numbers))
print("Slicing")
print(numbers[0:4])
print(numbers[2:5])
print(numbers[5:8])
print("Reverse order")
print(numbers[::-1])
print("Count")
a=int(input("Enter a number:"))
print(numbers.count(a))
print(numbers.index(a))
