class Animal:
    def sound(self):
        print("Animal makes sound")
class Dog:
    def sound(self):
        print("Dog barks")
class Cat:
    def sound(self):
        print("Cat says meow")
cat1=Cat()
dog1=Dog()
animal1=Animal()
animals=[animal1,dog1,cat1]
for animal in animals:
    animal.sound()
