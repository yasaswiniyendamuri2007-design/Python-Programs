class Animal:
    def sound(self):
        print("Animal makes a sound")
class Cat(Animal):
    def sound(self):
        print("Cat says meow")
class Dog(Animal):
    def sound(self):
        print("Dog barks")
def make_sound(animal):
    animal.sound()
animal1=Animal()
dog1=Dog()
cat1=Cat()
make_sound(dog1)
make_sound(animal1)
make_sound(cat1)
