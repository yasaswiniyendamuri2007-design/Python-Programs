class Student:
    def __init__(self,marks):
        self.__marks=marks
    def get_marks(self):
        return self.__marks
    def set_marks(self,marks):
        if 0<=marks<=100:
            self.__marks=marks
            print("Marks updated successfully")
        else:
            print("Invalid marks")
student1=Student(80)
print("Name:",student1.get_marks())
student1.set_marks(120)
print("After updation:",student1.get_marks())
