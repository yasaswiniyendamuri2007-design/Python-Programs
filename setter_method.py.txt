class Student:
    def __init__(self,name):
        self.__name=name
    def get_name(self):
        return self.__name
    def set_name(self,name):
        self.__name=name
student1=Student("Yasaswini")
print("Name:",student1.get_name())
student1.set_name("Rahul")
print("Name after updation:",student1.get_name())
