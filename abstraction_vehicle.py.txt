from abc import ABC,abstractmethod
class Vehicle(ABC):
    @abstractmethod
    def start(self):
        pass
class Car(Vehicle):
    def start(self):
        print("Car starts with a key")
class Bike(Vehicle):
    def start(self):
        print("Bike starts with a button")
class Bus(Vehicle):
    def start(self):
        print("Bus starts with a key")
car1=Car()
bike1=Bike()
bus1=Bus()
car1.start()
bike1.start()
bus1.start()
